# WakeUp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A native macOS app that prevents sleep via IOKit power assertions and covers all displays with a fake macOS-style clock screensaver, exiting on any mouse/keyboard activity.

**Architecture:** SwiftPM executable target using SwiftUI. Four modules: `SleepPreventer` (IOPM assertions), `ScreensaverOverlay` (borderless fullscreen NSWindows per screen), `ExitMonitor` (NSEvent monitors with a 10px mouse-move threshold), and `SessionController` (orchestrates start/countdown/stop, drives the SwiftUI main window). A shell script bundles the release binary into `WakeUp.app`.

**Tech Stack:** Swift 5.9+, SwiftUI, AppKit, IOKit.pwr_mgt, XCTest via `swift test`. No third-party dependencies.

## Global Constraints

- Platform: macOS 13+ (`platforms: [.macOS(.v13)]` in Package.swift).
- No third-party dependencies — Apple frameworks only.
- Spec: `docs/superpowers/specs/2026-07-16-wakeup-design.md`.
- Power assertion reason string is exactly `"WakeUp working mode"` (tests grep for it in `pmset` output).
- Mouse-move exit threshold: 10pt accumulated distance. Clicks/keys exit immediately.
- UI copy is Simplified Chinese, per the spec (e.g. 「开始工作模式」).
- Tests must not open visible windows or fullscreen overlays (only pure logic + IOPM assertions are unit-tested; window/event behavior is verified manually in Task 8).

import Foundation

enum ClockFormatter {
    static func timeString(from date: Date, timeZone: TimeZone = .current) -> String {
        let f = DateFormatter()
        f.timeZone = timeZone
        f.dateFormat = "HH:mm"
        return f.string(from: date)
    }

    static func dateString(from date: Date,
                           locale: Locale = .current,
                           timeZone: TimeZone = .current) -> String {
        let f = DateFormatter()
        f.locale = locale
        f.timeZone = timeZone
        f.dateFormat = DateFormatter.dateFormat(fromTemplate: "EEEEMMMd", options: 0, locale: locale)
        return f.string(from: date)
    }
}

enum DurationFormatter {
    static func string(from interval: TimeInterval) -> String {
        let total = Int(interval)
        let h = total / 3600
        let m = (total % 3600) / 60
        if h > 0 { return "\(h) 小时 \(m) 分钟" }
        if m > 0 { return "\(m) 分钟" }
        return "不到 1 分钟"
    }
}

import Foundation

struct MouseMoveAccumulator {
    private(set) var accumulated: CGFloat = 0
    let threshold: CGFloat

    init(threshold: CGFloat = 10) {
        self.threshold = threshold
    }

    mutating func add(dx: CGFloat, dy: CGFloat) -> Bool {
        accumulated += (dx * dx + dy * dy).squareRoot()
        return accumulated >= threshold
    }
}
```

import Foundation
import IOKit.pwr_mgt

final class SleepPreventer {
    private var systemAssertionID: IOPMAssertionID = 0
    private var displayAssertionID: IOPMAssertionID = 0
    private(set) var isActive = false

    private static let reason = "WakeUp working mode" as CFString

    func start() {
        guard !isActive else { return }
        IOPMAssertionCreateWithName(
            kIOPMAssertionTypePreventUserIdleSystemSleep as CFString,
            IOPMAssertionLevel(kIOPMAssertionLevelOn),
            Self.reason,
            &systemAssertionID
        )
        IOPMAssertionCreateWithName(
            kIOPMAssertionTypePreventUserIdleDisplaySleep as CFString,
            IOPMAssertionLevel(kIOPMAssertionLevelOn),
            Self.reason,
            &displayAssertionID
        )
        isActive = true
    }

    func stop() {
        guard isActive else { return }
        IOPMAssertionRelease(systemAssertionID)
        IOPMAssertionRelease(displayAssertionID)
        systemAssertionID = 0
        displayAssertionID = 0
        isActive = false
    }

    deinit {
        stop()
    }
}
```

struct ScreensaverView: View {

    @State private var drift = CGSize.zero

    private let timer = Timer.publish(every: 1, on: .main, in: .common).autoconnect()

    var body: some View {

            now = date
            if minuteChanged {
                withAnimation(.easeInOut(duration: 2)) {
                    drift = CGSize(width: CGFloat.random(in: -30...30),

- [ ] **Step 2: Write ScreensaverOverlay.swift**

Borderless windows cannot become key by default; the `OverlayWindow` subclass overrides `canBecomeKey` so the local key-event monitor (Task 6) receives keystrokes without any accessibility permission.

```swift
import AppKit
import SwiftUI

final class OverlayWindow: NSWindow {
    override var canBecomeKey: Bool { true }
}

final class ScreensaverOverlay {
    private var windows: [NSWindow] = []
    private var screenObserver: Any?
    private(set) var isShowing = false

    func show() {
        guard !isShowing else { return }
        isShowing = true
        buildWindows()
        NSCursor.hide()
        NSApp.activate(ignoringOtherApps: true)
        screenObserver = NotificationCenter.default.addObserver(
            forName: NSApplication.didChangeScreenParametersNotification,
            object: nil, queue: .main
        ) { [weak self] _ in
            guard let self, self.isShowing else { return }
            self.tearDownWindows()
            self.buildWindows()
        }
    }

    func hide() {
        guard isShowing else { return }
        isShowing = false
        if let observer = screenObserver {
            NotificationCenter.default.removeObserver(observer)
            screenObserver = nil
        }
        tearDownWindows()
        NSCursor.unhide()
    }

    private func buildWindows() {
        for (index, screen) in NSScreen.screens.enumerated() {
            let window = OverlayWindow(
                contentRect: screen.frame,
                styleMask: [.borderless],
                backing: .buffered,
                defer: false
            )
            window.isReleasedWhenClosed = false
            window.level = .screenSaver
            window.isOpaque = true
            window.backgroundColor = .black
            window.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .stationary]
            window.contentView = NSHostingView(rootView: ScreensaverView())
            window.setFrame(screen.frame, display: true)
            if index == 0 {
                window.makeKeyAndOrderFront(nil)
            } else {
                window.orderFrontRegardless()
            }
            windows.append(window)
        }
    }

    private func tearDownWindows() {
        for window in windows {
            window.orderOut(nil)
        }
        windows.removeAll()
    }
}
```

### Task 6: ExitMonitor

**Files:**
- Create: `Sources/WakeUp/ExitMonitor.swift`

**Interfaces:**
- Consumes: `MouseMoveAccumulator` from Task 3 (threshold logic already unit-tested there).
- Produces: `final class ExitMonitor { func arm(onExit: @escaping () -> Void); func disarm() }`. Used by `SessionController` (Task 7). Fires `onExit` exactly once per arm, then self-disarms.

Permissions note: global keyboard monitoring requires Accessibility permission, which we deliberately avoid. Keyboard exit is handled by the **local** monitor — it works because the overlay window is key and our app is frontmost (Task 5). The **global** monitor covers mouse events (no permission needed), as a safety net if focus is lost.

- [ ] **Step 1: Write ExitMonitor.swift**

```swift
import AppKit

final class ExitMonitor {
    private var globalMonitor: Any?
    private var localMonitor: Any?
    private var accumulator = MouseMoveAccumulator()
    private var onExit: (() -> Void)?

    func arm(onExit: @escaping () -> Void) {
        disarm()
        self.onExit = onExit
        accumulator = MouseMoveAccumulator()

        let mask: NSEvent.EventTypeMask = [
            .mouseMoved, .leftMouseDragged, .rightMouseDragged,
            .leftMouseDown, .rightMouseDown, .otherMouseDown,
            .scrollWheel, .keyDown
        ]
        globalMonitor = NSEvent.addGlobalMonitorForEvents(matching: mask) { [weak self] event in
            self?.handle(event)
        }
        localMonitor = NSEvent.addLocalMonitorForEvents(matching: mask) { [weak self] event in
            self?.handle(event)
            return event
        }
    }

    func disarm() {
        if let monitor = globalMonitor { NSEvent.removeMonitor(monitor) }
        if let monitor = localMonitor { NSEvent.removeMonitor(monitor) }
        globalMonitor = nil
        localMonitor = nil
        onExit = nil
    }

    private func handle(_ event: NSEvent) {
        switch event.type {
        case .mouseMoved, .leftMouseDragged, .rightMouseDragged:
            if accumulator.add(dx: event.deltaX, dy: event.deltaY) {
                fire()
            }
        default:
            fire()
        }
    }

    private func fire() {
        let callback = onExit
        disarm()
        callback?()
    }
}
```

```swift
import AppKit
import Combine

@MainActor
final class SessionController: ObservableObject {
    enum State: Equatable {
        case idle
        case countingDown(remaining: Int)
        case active
    }

    @Published private(set) var state: State = .idle
    @Published private(set) var lastSessionSummary: String?

    private let sleepPreventer = SleepPreventer()
    private let overlay = ScreensaverOverlay()
    private let exitMonitor = ExitMonitor()
    private var countdownTimer: Timer?
    private var sessionStart: Date?

    func startWorkMode(delaySeconds: Int) {
        guard state == .idle else { return }
        guard delaySeconds > 0 else {
            activate()
            return
        }
        state = .countingDown(remaining: delaySeconds)
        countdownTimer = Timer.scheduledTimer(withTimeInterval: 1, repeats: true) { [weak self] timer in
            Task { @MainActor in
                guard let self, case .countingDown(let remaining) = self.state else {
                    timer.invalidate()
                    return
                }
                if remaining <= 1 {
                    timer.invalidate()
                    self.activate()
                } else {
                    self.state = .countingDown(remaining: remaining - 1)
                }
            }
        }
    }

    func cancelCountdown() {
        guard case .countingDown = state else { return }
        countdownTimer?.invalidate()
        countdownTimer = nil
        state = .idle
    }

    private func activate() {
        countdownTimer?.invalidate()
        countdownTimer = nil
        sessionStart = Date()
        sleepPreventer.start()
        overlay.show()
        exitMonitor.arm { [weak self] in
            Task { @MainActor in self?.deactivate() }
        }
        state = .active
    }

    private func deactivate() {
        guard state == .active else { return }
        exitMonitor.disarm()
        overlay.hide()
        sleepPreventer.stop()
        if let start = sessionStart {
            let duration = DurationFormatter.string(from: Date().timeIntervalSince(start))
            lastSessionSummary = "已结束，本次持续 \(duration)"
        }
        sessionStart = nil
        state = .idle
    }
}
```

# WakeUp

macOS 防休眠 + 假屏保工具：一键进入「工作模式」——阻止电脑休眠，同时全屏显示
仿 macOS 时钟屏保，让 AI agent（如 Claude Code）在后台安心干活。

## 使用

1. 构建：`./scripts/make-app.sh`，产物在 `build/WakeUp.app`，双击运行。
2. 点「开始工作模式」，倒计时结束后进入全屏时钟屏保，并开始阻止系统休眠。
3. 动一下鼠标（约 10pt 以上）或按任意键即退出屏保，恢复正常电源策略。

## 注意事项

- **合盖仍会休眠**：软件层面的电源断言无法阻止合盖睡眠，请保持屏幕打开。
- 屏保遮盖所有显示器，覆盖 Dock 与菜单栏；插拔显示器会自动适配。
- App 退出（含崩溃）时电源断言自动释放，不会残留影响系统休眠策略。
- 无需辅助功能权限：键盘退出依赖本 App 处于前台（屏保激活时即是）。

## 开发

- `swift build` 构建，`swift test` 跑测试。
- 验证电源断言：工作模式激活时运行 `pmset -g assertions`，
  应看到两条名为 "WakeUp working mode" 的断言。
